사진 데이터는 \예배부 데이터\사진에, 출석데이터는 \예배부 데이터\출석에 각각 자동 분류되어 저장됩니다.

                

                
            


            

                
1Google Apps Script 코드 배포

                
구글 스프레드시트의 [확장 프로그램] → [Apps Script] 또는 script.google.com에서
                    새 스크립트를 열고 아래 코드를 그대로 붙여넣습니다.

                

                    
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // 0. 사진 삭제 요청 처리
    if (data.action === 'delete' || data.action === 'deletePhoto') {
      var ok = deleteDrivePhoto(data.fileId, data.fileName);
      return ContentService.createTextOutput(JSON.stringify({
        success: ok,
        fileId: data.fileId,
        fileName: data.fileName,
        message: ok ? '구글 드라이브에서 삭제되었습니다.' : '파일을 찾을 수 없습니다.'
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // 1. 최상위 루트 폴더: "예배부 데이터"
    var rootFolderName = "예배부 데이터";
    var rootFolders = DriveApp.getFoldersByName(rootFolderName);
    var rootFolder = rootFolders.hasNext() ? rootFolders.next() : DriveApp.createFolder(rootFolderName);
    
    // 2. 하위 폴더: "사진" 또는 "출석"
    var subFolderName = (data.type === 'photo' || data.subFolder === '사진' || (!data.text && data.base64)) ? "사진" : "출석";
    if (data.subFolder) subFolderName = data.subFolder;
    
    var subFolders = rootFolder.getFoldersByName(subFolderName);
    var targetFolder = subFolders.hasNext() ? subFolders.next() : rootFolder.createFolder(subFolderName);
    
    var fileName = data.fileName || ('예배부_' + Utilities.formatDate(new Date(), 'GMT+9', 'yyyyMMdd_HHmmss') + (subFolderName === '사진' ? '.jpg' : '.txt'));
    var file;
    
    // 3. 파일 생성 (출석 텍스트 or 사진 바이너리)
    if (data.text) {
      file = targetFolder.createFile(fileName, data.text, MimeType.PLAIN_TEXT);
    } else if (data.base64) {
      var base64Data = data.base64;
      if (base64Data.indexOf('base64,') > -1) {
        base64Data = base64Data.split('base64,')[1];
      }
      var decoded = Utilities.base64Decode(base64Data);
      var blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', fileName);
      file = targetFolder.createFile(blob);
    } else {
      throw new Error('전송할 데이터가 없습니다.');
    }
    
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      fileId: file.getId(),
      fileName: file.getName(),
      viewUrl: file.getUrl(),
      savedPath: '\\예배부 데이터\\' + subFolderName
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var action = (e && e.parameter) ? e.parameter.action : '';
  
  // 1. 사진 삭제 요청 (GET 방식)
  if (action === 'delete' || action === 'deletePhoto') {
    var fileId = e.parameter.fileId;
    var fileName = e.parameter.fileName;
    var ok = deleteDrivePhoto(fileId, fileName);
    return ContentService.createTextOutput(JSON.stringify({
      success: ok,
      fileId: fileId,
      fileName: fileName,
      message: ok ? '구글 드라이브에서 삭제되었습니다.' : '파일을 찾을 수 없습니다.'
    })).setMimeType(ContentService.MimeType.JSON);
  }
  
  // 2. 사진 목록 조회 (실시간 동기화)
  if (action === 'getPhotos') {
    try {
      var rootFolders = DriveApp.getFoldersByName("예배부 데이터");
      if (!rootFolders.hasNext()) {
        return ContentService.createTextOutput(JSON.stringify({ success: true, photos: [] })).setMimeType(ContentService.MimeType.JSON);
      }
      var rootFolder = rootFolders.next();
      var photoFolders = rootFolder.getFoldersByName("사진");
      if (!photoFolders.hasNext()) {
        return ContentService.createTextOutput(JSON.stringify({ success: true, photos: [] })).setMimeType(ContentService.MimeType.JSON);
      }
      var photoFolder = photoFolders.next();
      var files = photoFolder.getFiles();
      var photos = [];
      while (files.hasNext()) {
        var file = files.next();
        if (file.isTrashed()) continue; // 휴지통 파일 제외
        photos.push({
          id: file.getId(),
          fileName: file.getName(),
          driveUrl: file.getUrl(),
          thumbUrl: 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w600',
          createdAt: file.getDateCreated().toISOString()
        });
      }
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        photos: photos
      })).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        error: err.toString()
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  // 3. 상태 확인
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    folder: '예배부 데이터',
    subFolders: ['사진', '출석'],
    message: '성광교회 예배부 데이터 저장소 (사진: \\예배부 데이터\\사진, 출석: \\예배부 데이터\\출석) 준비 완료'
  })).setMimeType(ContentService.MimeType.JSON);
}

// 구글 드라이브 사진 파일 삭제(휴지통 이동) 함수
function deleteDrivePhoto(fileId, fileName) {
  var deleted = false;
  if (fileId) {
    try {
      var file = DriveApp.getFileById(fileId);
      file.setTrashed(true);
      deleted = true;
    } catch (e) {}
  }
  if (!deleted && fileName) {
    try {
      var rootFolders = DriveApp.getFoldersByName("예배부 데이터");
      if (rootFolders.hasNext()) {
        var rootFolder = rootFolders.next();
        var photoFolders = rootFolder.getFoldersByName("사진");
        if (photoFolders.hasNext()) {
          var photoFolder = photoFolders.next();
          var files = photoFolder.getFilesByName(fileName);
          while (files.hasNext()) {
            var f = files.next();
            f.setTrashed(true);
            deleted = true;
          }
        }
      }
    } catch (e) {}
  }
  return deleted;
}
